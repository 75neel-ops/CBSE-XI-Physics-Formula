def study_mode():
    print("\n--- 📚 FORMULA CHEAT SHEET ---")
    for topic, items in FORMULAS.items():
        print(f"\n🔹 {topic}:")
        for item in items:
            print(f"  • {item['prompt']}  ➡️  {item['answer']}")

def quiz_game():
    print("\n--- 🎮 FORMULA MASTERY GAME ---")
    score = 0
    total = 0
    
    # Flatten formula list for random quiz questions
    all_questions = []
    for topic, items in FORMULAS.items():
        for item in items:
            all_questions.append((topic, item))
            
    random.shuffle(all_questions)
    
    for topic, item in all_questions[:5]:  # Ask 5 random questions
        print(f"\n[Topic: {topic}]")
        user_ans = input(f"Fill in the blank: {item['prompt']} = ").strip()
        
        if user_ans.lower().replace(" ", "") == item['answer'].lower().replace(" ", ""):
            print("✅ Correct!")
            score += 1
        else:
            print(f"❌ Incorrect! Correct answer: {item['answer']}")
        total += 1
        
    print(f"\n🏆 Game Over! Your Score: {score}/{total}")

def main():
    while True:
        print("\n===============================")
        print("  CLASS 11 MATH FORMULA GAME   ")
        print("===============================")
        print("1. Study All Formulas")
        print("2. Play Formula Quiz Game")
        print("3. Exit")
        
        choice = input("\nChoose an option (1-3): ").strip()
        if choice == "1":
            study_mode()
        elif choice == "2":
            quiz_game()
        elif choice == "3":
            print("Good luck with your Class 11 Maths studies! Goodbye!")
            break
        else:
            print("Invalid option, try again.")

if __name__ == "__main__":
    main()
